cmake_minimum_required(VERSION 3.22.1)
project(vehicle_someip LANGUAGES CXX)

# The host supplies vsomeip3 (imported Android artifact or system package).
if(NOT TARGET vsomeip3)
    find_package(vsomeip3 REQUIRED)
endif()
add_library(vehicle_someip STATIC someip/SomeIpClient.cpp)
set_target_properties(vehicle_someip PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_compile_features(vehicle_someip PUBLIC cxx_std_17)
target_include_directories(vehicle_someip PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
if(VSOMEIP_SOURCE_ROOT)
    target_include_directories(vehicle_someip PRIVATE "${VSOMEIP_SOURCE_ROOT}/interface")
endif()
target_link_libraries(vehicle_someip PUBLIC vsomeip3)
if(ANDROID)
    target_link_libraries(vehicle_someip PRIVATE log)
else()
    find_package(Threads REQUIRED)
    target_link_libraries(vehicle_someip PRIVATE Threads::Threads)
endif()
