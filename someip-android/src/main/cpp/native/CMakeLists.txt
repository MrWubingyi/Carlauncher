cmake_minimum_required(VERSION 3.22.1)
project(someip_client LANGUAGES CXX)

# The host supplies vsomeip3 (imported Android artifact or system package).
if(NOT TARGET vsomeip3)
    find_package(vsomeip3 REQUIRED)
endif()
add_library(someip_client STATIC someip/SomeIpClient.cpp)
set_target_properties(someip_client PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_compile_features(someip_client PUBLIC cxx_std_17)
target_include_directories(someip_client PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
if(VSOMEIP_SOURCE_ROOT)
    target_include_directories(someip_client PRIVATE "${VSOMEIP_SOURCE_ROOT}/interface")
endif()
target_link_libraries(someip_client PUBLIC vsomeip3)
if(ANDROID)
    target_link_libraries(someip_client PRIVATE log)
else()
    find_package(Threads REQUIRED)
    target_link_libraries(someip_client PRIVATE Threads::Threads)
endif()
